\subsection{微分積分}
\prob{
  $x$と$y$の関数
  \begin{gather}
    f = | 4y^2 - 4xy + 2x^2 - 2x + 1 | + | 2x |
  \end{gather}
  に関する次の問いに答えなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item この関数を絶対値記号を用いない形で表しなさい。
    \item この関数の最小値とそれを与える$x$と$y$の値を求めなさい。
  \end{enumerate}
}

\prob{
  $x$と$y$が
  \begin{gather}
    x^4 - 4 x^2 y +y^3 = 0,\,y > 0
  \end{gather}
  という関係式を満たすときに、下記の問いに答えなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item 変数$t$を
    \begin{gather}
      t = \frac{x^2}{y}
    \end{gather}
    とおき、この関係式(1)を$t$と$y$に関する方程式に書き直しなさい。
    \item 次の極限値を求めなさい。
    \begin{gather}
      \lim_{y \to 0} \frac{x^2}{y}
    \end{gather}
  \end{enumerate}
}

\prob{
  次の不定積分を求めなさい。
  \begin{gather}
    \int \frac{16x^3}{x^8 - 4} dx
  \end{gather}
}

\prob{
  次の微分方程式を解きなさい。
  \begin{gather}
    \frac{d^3 y}{dx^3} - \frac{d^2 y}{dx^2} - 4\frac{dy}{dx} + 4y = x - 1
  \end{gather}
}

\subsection{線形代数}

\prob{
  下記に示す3次の正方行列について、固有値ならびに固有ベクトルを全て求めよ。
  \begin{gather}
    \bA = \begin{pmatrix}
      1 & -1 & 4 \\
      3 & 2 & -1 \\
      2 & 1 & -1
    \end{pmatrix}
  \end{gather}
}

\prob{
  3次の正方行列$\bA,\,\bB$に関する下記のそれぞれの式について、任意の行列に対して成立するか否かを示せ。
  その際、成分表示が必要であれば、$a_{ij},\,b_{ij}\,(i,\,j=1,\,2,\,3)$などとせよ。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item $\bA\bB = \bB\bA$
    \item $(\bA\bB)^{t} = \bB^t \bA^t$
  \end{enumerate}
}

\prob{
  下記の連立一次方程式を係数行列を用いて表示し、掃き出し法により解け。
  その際、拡大係数行列の形式で表示しても良い。
  \begin{gather}
    3x + 6y - 5z = -3 \\
    x - 2y + z = 3 \\
    5x - 7y + 2z = 9
  \end{gather}
}
