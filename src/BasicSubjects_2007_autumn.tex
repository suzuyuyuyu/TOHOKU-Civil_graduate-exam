\section{2007秋}
\subsection{微分積分}
\prob{
  次の関数を$x$について微分しなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp x^{\log x}$
    \item $\disp \frac{1 - e^{x}}{1 + e^{x}}$
  \end{enumerate}
}

\prob{
  次の定積分を求めなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp \int_{0}^{\infty} a^{x} dx  \quad (0 < a < 1)$
    \item $\disp \int_{0}^{1} x e^{x} dx$
  \end{enumerate}
}

\prob{
  次の微分方程式を解きなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp \frac{dy}{dx} + \frac{1}{x}y = \frac{1}{x}$
    \item $\disp \frac{d^2y}{dx^2} - y = x$
  \end{enumerate}
}

\subsection{線形代数}
\prob{
  行列$\disp \bA = \begin{bmatrix}
    4 & -1 \\
    2 & 1
  \end{bmatrix}$について、
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item $\bA$の固有値と固有ベクトルを求めなさい。
    \item $\bA$の2つの固有ベクトルは線形独立であることを示しなさい。
    \item 正則行列$\bP$を用いて$\bA$を対角化しなさい。
  \end{enumerate}
}

\prob{
  平面$\disp \bQ = \Dset{ \begin{bmatrix}
    x \\ y \\ z
  \end{bmatrix} \relmiddle 2x + y + z = 0 }$を張る基底を求めなさい。
}

\prob{
  次の連立方程式の一般解を求めなさい。
  \begin{gather}
    \left\{
    \begin{aligned}
      x_{1} + x_{2} + x_{3} + x_{4} = 0 \\
      x_{1} + x_{2} + 3x_{3} + 4x_{4} = 0 \\
      2x_{1} + 2x_{2} + 4x_{3} + 5x_{4} = 0
    \end{aligned}
    \right.
  \end{gather}
}
