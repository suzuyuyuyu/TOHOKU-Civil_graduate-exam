\subsection{微分積分}

\prob{
  次の極限値を求めなさい。
  \begin{gather}
    \lim_{x\to 0} \frac{1}{x^2} (e^{-x} + e^{-2x})(1 - \cos x)
  \end{gather}
}

\prob{
  関数
  \begin{gather}
    f(x,y) = (x + ay) \exp(-x^2 - 2y^2)
  \end{gather}
  に対して（$a$は定数）、
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp \frac{\pd f}{\pd x}$と$\disp \frac{\pd f}{\pd y}$を求めなさい。
    \item $a = 1$とするとき、$f$の極値を求めなさい。
  \end{enumerate}
}

\prob{
  ${}$
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item 下記の領域$D$を図示しなさい。
    \begin{gather}
      D = \Dset{(x,y) | x^2 + \frac{y^2}{4} \leq 1,\, 2x + y \geq 2}
    \end{gather}
    \item 下記の定積分を求めなさい。
    \begin{gather}
      \iint_{D} y \,dydx
    \end{gather}
  \end{enumerate}
}

\prob{
  微分方程式
  \begin{gather}
    \frac{d^4 y}{dx^4} + \frac{d^3 y}{dx^3} + \frac{d y}{dx} - y = x + 4 + e^x
  \end{gather}
  を解きなさい。
}

\subsection{線形代数}

\prob{
  行列$\bA$が
  \begin{gather}
    \bA \equiv \pmat{
      3 & 1 & -3 \\
      1 & 3 & -3 \\
      1 & 1 & -1
    }
  \end{gather}
  と定義されている。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item この行列の固有値を求めよ。
    \item この行列の対角化が可能な場合には対角化せよ。
  \end{enumerate}
}

\prob{
  行列$\bB$が
  \begin{gather}
    \bB \equiv \pmat{
      1 & 0 & 2 \\
      1 & 2 & 1 \\
      1 & 1 & 1
    }
  \end{gather}
  と定義されている。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item この行列$\bB$の逆行列を求めよ。
    \item 未知数$\bu$に対する次の連立方程式を考える。
    \begin{gather}
      \bB \bu = \bf,\,
      \bu = \pmat{x \\ y \\ z},\,
      \bf = \pmat{1 \\ a \\ 2}
    \end{gather}
    この方程式の解$\bu$は$\bf$と直行する。$a$を求めよ。
  \end{enumerate}

}
