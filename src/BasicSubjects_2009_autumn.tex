\subsection{微分積分}
\prob{
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item 関数$y = (\sin x)^x$について、$\disp \frac{dy}{dx}$を求めなさい。
    \item 関数$\disp f(x,y) = \frac{\sin^{-1} \biggl(\cfrac{x}{y}\biggr) }{x^y}$について、
    $\disp \frac{\pd f}{\pd x}$および$\disp \frac{\pd f}{\pd y}$を求めなさい。
  \end{enumerate}
}


\prob{
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item 次の不定積分を求めなさい。
    \begin{gather}
      \int \frac{x + 2}{x(x + 1)^2} dx
    \end{gather}
    \item 次の定積分を求めなさい。
    \begin{gather}
      \iint_{d} x\cdot e^{x^2 + y} dxdy,\quad \text{但し、} D = \Dset{ (x,y) \relmiddle 0 \leq x \leq 1,\,0\leq y \leq x^2 }
    \end{gather}
  \end{enumerate}
}


\prob{
  次の微分方程式を解きなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp x^2 \frac{dy}{dx} + y = 1$
    \item $\disp \frac{d^2 y}{dx^2} - 3 \frac{dy}{dx} + 2y = e^{3x}$
  \end{enumerate}
}

\subsection{線形代数}
\prob{
  ベクトル$\ba = (a_1,\,a_2,\,a_3)$、$\bb = (b_1,\,b_2,\,b_3)$、$\bc = (c_1,\,c_2,\,c_3)$を
  隣り合う3辺とする平行六面体の体積$V$に関する以下の問に答えなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item ベクトルの内積と外積を用いて、この体積$V$を表しなさい。
    \item 行列式を用いて、この体積$V$を表しなさい。
    \item $\ba = (1,2,1)$、$\bb = (0,2,3)$、$\bc = (0,-1,-2)$と与えるとき、この体積$V$を求めなさい。
  \end{enumerate}
}

\prob{
  以下の行列の固有値および固有ベクトルを求めなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item${}$
    \begin{gather}
      \begin{bmatrix}
        1 & 0 & 1 \\
        0 & 1 & 1 \\
        0 & 0 & 1
      \end{bmatrix}
    \end{gather}
    \item${}$
    \begin{gather}
      \begin{bmatrix}
        2 & 1 & 0 \\
        1 & 2 & 0 \\
        0 & 1 & 0
      \end{bmatrix}
    \end{gather}
  \end{enumerate}
}
