\section{2002秋}
\subsection{微分積分}
\prob{
  次の式を$x$について微分せよ。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item $\disp\ln \frac{2x + 1}{x - 2}$
    \item $x^{x}$
    \item $\arctan x$\qquad （$\arctan\colon \tan\text{の逆関数}$）
  \end{enumerate}
}

\prob{
  次の定積分を求めよ。必要ならば
  \begin{gather}
    \int_{0}^{\infty} e^{-x^{2}}dx = \sqrt{\pi}/2,\quad \lim_{x\to\infty}x e^{-x^{2}} = 0
  \end{gather}
  の関係を用いよ。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp  \int_{0}^{\infty} e^{-4x^{2}} dx$
    \item $\disp  \int_{0}^{\infty} x e^{-4x^{2}} dx$
    \item $\disp  \int_{0}^{\infty} x^2 e^{-4x^{2}} dx$
  \end{enumerate}
}

\subsection{線形代数}
\prob{
  行列$\bA$および列ベクトル$\bx,\,\by,\,bb$が次式で表されるとする。以下の問に答えよ。
  \begin{gather}
    \bA = \begin{bmatrix}
      1 & -1 & k \\
      0 & 2 & k \\
      1 & 1 & -2
    \end{bmatrix},\,
    \bx = \begin{bmatrix}
      x_{1} \\ x_{2} \\ x_{3}
    \end{bmatrix},\,
    \by = \begin{bmatrix}
      y_{1} \\ y_{2} \\ y_{3}
    \end{bmatrix},\,
    \bb = \begin{bmatrix}
      2 \\ 1 \\ q
    \end{bmatrix}
  \end{gather}
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item \label{2002autumn_l1_1}$\bx$に関する方程式系$\bA\bx = 0$が自明でない（ゼロでない）解を持つとき、$k$を求めよ。
    \item \ref{2002autumn_l1_1}が成立するとする。
    $\by$に関する方程式系$\bA\by = \bb$が解を持つとき、$q$を求めよ。
  \end{enumerate}
}

\prob{
  次の行列$\bM$について以下の問に答えよ。
  \begin{gather}
    \bM = \begin{bmatrix}
      2 & 1 & 0 \\
      1 & 2 & 0 \\
      0 & 0 & 1
    \end{bmatrix}
  \end{gather}
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item \label{2002autumn_l2_1}
    行列$\bM$の固有値および対応する固有ベクトルを全て求めよ。
    \item \ref{2002autumn_l2_1}で求めた固有値および固有ベクトルを用いて、行列$\bM$を対角化せよ。
    \item 上の結果を用いて2次形式$Q = 2x^2 + 2y^2 + z^2 + 2xy$を標準形になおせ。
  \end{enumerate}
}
