\section{2000秋}
\subsection{微分積分}
\prob{
  微分せよ。
  \begin{gather}
    y(x) = \sin(\cos x) \label{eq:q-2000a-calculus-1}
  \end{gather}
}
\begin{ans*}
  \begin{gather}
    y'(x) = -\cos(\cos x)\sin x
  \end{gather}
\end{ans*}

\prob{
  関数$f(x,y)$が
  \begin{gather}
    \nabla^{2}f = 0,\quad \nabla^{2} = \frac{\pd^2}{\pd x^2} + \frac{\pd^2}{\pd y^2}
  \end{gather}
  を満たしているとき、次の値を求めよ。
  \begin{gather}
    \nabla^{4}(yf),\quad \nabla^{4} = \nabla^{2}\cdot \nabla^{2}
  \end{gather}
}

\prob{
  積分せよ。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp \int x\sqrt{4x^2 + 9}dx$
    \item $\disp \int \frac{1}{a^2 - x^2}dx$
  \end{enumerate}
}

\subsection{線形代数}
\prob{
  $n\tm n$の行列$[\bA]^{2} = [\bA][\bA]$の逆が$n\tm n$の行列$[\bB]$のとき、
  $[\bA]$の逆が$[\bA\bB]$であることを示せ。
}

\prob{
  ベクトル$[\bd]^{\rm{T}} = [1 ~ -1 ~ 1]$と次の$3\tm 3$の正方行列$[\bC]$について以下の問いに答えよ。
  \begin{gather}
    [\bC] = \begin{bmatrix}
      1 & 2 & 0 \\
      2 & 2 & 0 \\
      0 & 0 & 1
    \end{bmatrix}
  \end{gather}
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item 行列$[\bC]$の行列式の値（$\det([\bC])$）と階数（$\rank([\bC])$）を求めよ。
    \item 次の各演算を行え。
    \begin{enumerate}[label=(\alph*), ref=(\alph*), itemsep=0pt]
      \item $[\bd]^{\rm{T}}[\bd]$
      \item $[\bd][\bd]^{\rm{T}}$
      \item $([\bC][\bd])^{\rm{T}}[\bd]$
    \end{enumerate}
  \end{enumerate}
}

\prob{
  次の行列について以下の問いに答えよ。
  \begin{gather}
    [\bA] = \begin{bmatrix}
      1 & 0 & -1 \\
      0 & 1 & 1 \\
      0 & 0 & 2
    \end{bmatrix}
  \end{gather}
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item すべての固有値とその重複度を求めよ。
    \item 対応する固有ベクトルを求めよ。
    \item 固有値を用いてこの行列$[\bA]$を対角化せよ。
  \end{enumerate}
}
