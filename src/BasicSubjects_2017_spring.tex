\subsection{微分積分}
\prob{
  次の関数$f(x)$について、以下の問いに答えなさい。
  $e$は自然対数の底（ネイピア数）である。
  \begin{gather}
    f(x) =\Bigl\{ e^{(\sqrt{x+9} - 3)} \Bigr\}^{1/x}
  \end{gather}
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item $\log f(x)$を分数の形で表しなさい。ただし、$\log$は対数を表す。
    \item 次の極限値を求めなさい。
    \begin{gather}
      \lim_{x\to 0}f(x)
    \end{gather}
  \end{enumerate}
}

\prob{
  次の不定積分を求めなさい。
  \begin{gather}
    \int \frac{x(3x - 1)}{(x - 1)(x^2 + 1)}\,dx
  \end{gather}
}

\prob{
  次の重積分について、以下の問いに答えなさい。
  \begin{gather}
    \iint_{D} \frac{1}{\sqrt{x^2+y}}\,dxdy,\quad D\colon 0\leq y\leq x^2,\,1\leq x\leq 2
  \end{gather}
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item 積分領域$D$を図示しなさい。
    \item この重積分を計算しなさい。
  \end{enumerate}
}

\prob{
  次の微分方程式を解きなさい。
  \begin{gather}
    y'' - 4y' + 4y = 5
  \end{gather}
}

\subsection{線形代数}
\prob{
  行列
  \begin{gather}
    \bA = \pmat{
      a & b & c & b \\
      b & c & b & c \\
      c & b & c & b \\
      b & c & b & a
    }
  \end{gather}
  に関する下記の問いに答えよ。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item $a=c$の時に$\det \bA$を求めよ。
    \item $a\neq c$の時に$\det \bA$を求めよ。
  \end{enumerate}
}

\prob{
  パラメータ$\gra,\,\beta$を含む一次方程式
  \begin{gather}
    \pmat{
      \gra & 1 & 1 \\
      1 & \gra & 1 \\
      1 & 1 & \gra
    }\pmat{
      x \\ y \\ z
    } = \pmat{
      \beta \\ \beta \\ \beta
    }
  \end{gather}
  を$x,\,y,\,z$について解け。
}

\prob{
  行列
  \begin{gather}
    \bA = \pmat{
      4 & 0 & 3 \\
      1 & -1 & 1 \\
      3 & 1 & 2
    }
  \end{gather}
  に関する下記の問いに答えよ。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item 固有値を全て求めよ。
    \item ある一つの固有値の固有ベクトルを求めよ。
  \end{enumerate}
}
