\subsection{微分積分}
\prob{
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item $\disp f(x,y) = \frac{x - y}{x + y}$について、
    $\frac{\pd f}{\pd x}$、$\frac{\pd f}{\pd y}$、$\frac{\pd^2 f}{\pd x \pd y}$を求めなさい。
    \item $f(x) = \log x$を$x = 1$のまわりでテイラー級数に展開しなさい。
  \end{enumerate}
}

\prob{
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item 次の微分方程式を解きなさい。
    \begin{gather}
      \frac{d^2y}{dx^2} - 4y = e^{-x}
    \end{gather}
    \item 次の境界条件を満たす(1)の微分方程式の解を求めなさい。
    \begin{gather}
      y(0) = 0,\quad \frac{dy}{dx}(0) = \frac{1}{3}
    \end{gather}
  \end{enumerate}
}

\prob{
  次の定積分を求めなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp \int_{0}^{3} \Biggl(\int_{0}^{\log y} e^{x + y} dx\Biggr) dy$
    \item $\disp \int_{0}^{1} \frac{2+x}{1+x^2} dx$
  \end{enumerate}
}

\subsection{線形代数}
\prob{
  実正方行列$\bA$が
  \begin{gather}
    \bA = \begin{pmatrix}
      5 & 1 & 3 \\
      1 & 5 & 3 \\
      3 & 3 & 3
    \end{pmatrix}
  \end{gather}
  で与えられている。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item この行列$\bA$で表された連立一次方程式
    \begin{gather}
      \bA \begin{Bmatrix}
        x_1 \\ x_2 \\ x_3
      \end{Bmatrix}
      = \begin{Bmatrix}
        4 \\ C \\ 6
      \end{Bmatrix}
    \end{gather}
    の解が存在するためには、$C$はどういう値であればよいか。
    \item $C$が前問(1)で求めた値であるとき、この問題の解をひとつ求めよ。
  \end{enumerate}
}

\prob{
  実正方行列$\bB$が
  \begin{gather}
    \bB = \begin{pmatrix}
      2 & -2 & -1 \\
      1 & -1 & -1 \\
      -2 & 4 & 3
    \end{pmatrix}
  \end{gather}
  で与えられている。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item 行列$\bB$のすべての固有値と、線形独立な三つの固有ベクトルを求めよ。
    \item その結果を用いて、この行列$\bB$を対角化せよ。
  \end{enumerate}
}
