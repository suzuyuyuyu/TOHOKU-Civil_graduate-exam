\section{2010秋}
\subsection{微分積分}
\prob{
  次の関数の導関数を求めなさい。
  ただし、逆三角関数は主値をを考慮するものとする。
  \begin{gather}
    \frac{1}{2}\Bigl( x\sqrt{1-x^2} + \sin^{-1}x \Bigr)
  \end{gather}
}

\prob{
  次の極限値を求めなさい。
  \begin{gather}
    \lim_{x\to 0} \frac{1}{x} \biggl( \frac{1}{x} - \frac{1}{\sin x} \biggr)
  \end{gather}
}

\prob{
  次の定積分を求めなさい。
  \begin{gather}
    \int_{0}^{\infty} \frac{dx}{e^x + e^{-x}}
  \end{gather}
}

\prob{
  次の微分方程式を解きなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp \frac{dy}{dx} = \frac{x^2 + y^2}{2xy}$
    \item $\disp \frac{dy}{dx} - \frac{y}{x} = \frac{1}{x^2}$
  \end{enumerate}
}

\subsection{線形代数}
\prob{
  行列$\disp \bA = \begin{bmatrix}
    1 & -2 & 0 \\
    0 & 3 & -3 \\
    0 & -1 & 1
  \end{bmatrix}$
  の固有値と固有ベクトルを求めよ。
}


\prob{
  次の2次曲線の方程式について、以下の問いに答えなさい。
  \begin{gather}
    x^2 - 4xy - 2y^2 + 12 = 0
  \end{gather}
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item この方程式を2次形式で表しなさい。
    \item 標準形を求めなさい。
    \item この方程式が示す曲線を描きなさい。
  \end{enumerate}
}

\prob{
  2次正方行列$\bA = \begin{bmatrix}
    a & b \\
    c & d
  \end{bmatrix}$
  について、次の問いに答えなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item 次の式の値を求めなさい。
    \begin{gather}
      \bA^2 - (a + d)\bA + (ad - bc)\bE
    \end{gather}
    ここで、$\disp \bE = \begin{bmatrix}
      1 & 0 \\
      0 & 1
    \end{bmatrix}$とする。
    \item $\bA^2 = \bA$を満足する2次正方行列$\bA$を求めなさい。
  \end{enumerate}
}
