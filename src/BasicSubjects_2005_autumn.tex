\subsection{微分積分}
\prob{
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp f(x,y) = \log \frac{1}{\sqrt{x^2 + 2y^3}}$について、$\disp \frac{\pd f}{\pd x}$および$\disp \frac{\pd f}{\pd y}$を求めなさい。
    \item $g(x,y,z)=\sin^{-1}\bigl(xy^4z^2\bigr)$について、$\disp \frac{\pd g}{\pd x}$、$\disp \frac{\pd g}{\pd y}$および$\disp \frac{\pd g}{\pd z}$を求めなさい。
  \end{enumerate}
}

\subsection{線形代数}
\prob{
  下記の連立方程式を解きなさい。
  \begin{gather}
    \begin{pmatrix}
      1 & -3 \\
      3 & -9
    \end{pmatrix}
    \begin{pmatrix}
      x \\ y
    \end{pmatrix}
    = \begin{pmatrix}
      1 \\ 3
    \end{pmatrix}
  \end{gather}
}

\prob{
  行列
  \begin{gather}
    \bA = \begin{pmatrix}
      a & b \\
      c & d
    \end{pmatrix}
  \end{gather}
  の固有値が$5$と$-3$であり、対応する固有値ベクトルがそれぞれ
  \begin{gather}
    \begin{pmatrix}
      7 \\ 1
    \end{pmatrix}
    ,\quad \begin{pmatrix}
      1 \\ -1
    \end{pmatrix}
  \end{gather}
  である。行列$\bA$を求めなさい。
}

\prob{
  下記の行列の行列式を求めなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp \bB = \begin{pmatrix}
      a & b & b \\
      b & a & b \\
      b & b & a
    \end{pmatrix}$
    \item $\disp \bC = \begin{pmatrix}
      a & b & b & b \\
      b & a & b & b \\
      b & b & a & b \\
      b & b & b & a
    \end{pmatrix}$
  \end{enumerate}
}
