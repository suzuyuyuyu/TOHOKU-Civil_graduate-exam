\subsection{微分積分}
\prob{
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=0pt]
    \item 次の関数を$x$で微分しなさい。
    \begin{gather}
      \sqrt{\frac{a^2 + x^2}{a^2 - x^2}}
    \end{gather}
    \item 次の関係があるとき、$\disp \frac{d^2 y}{dx^2}$を$x$と$y$で表しなさい。
    \begin{gather}
      x = \sin^2 (x + y),\quad |x| < 1.
    \end{gather}
  \end{enumerate}
}

\prob{
  次の積分を求めなさい。
  \begin{enumerate}[label=(\arabic*), ref=(\arabic*), itemsep=5pt]
    \item $\disp \int \frac{\bigl( a + b\sqrt{x + 1} \bigr)^2}{\sqrt{x + 1}} dx$
    \item $\disp \int x \ln \sqrt{x} dx$
  \end{enumerate}
}

\prob{
  次の微分方程式を解きなさい。
  \begin{gather}
    \frac{d^3y}{dx^3} - 3\frac{dy}{dx} + 2y = 5e^{3x}
  \end{gather}
}
